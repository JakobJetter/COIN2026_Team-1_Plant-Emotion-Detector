% ================= COINs / Springer T1-book look (pdflatex) =================
% NOTE: geometry (margins) is configured in report.qmd via the `geometry:` key,
% and graphicx/hyperref are provided by Quarto, so they are NOT loaded here
% (loading them twice causes an "option clash" error).

% ---- Times body via mathptmx (portable across all TeX Live installs) ----
\usepackage{mathptmx}

% ---- Paragraphs: 0.42 cm first-line indent, no inter-paragraph gap.
%      Requires `indent: true` in report.qmd (otherwise Quarto zeroes the indent).
%      article does not indent the first paragraph after a heading -> matches template.
\setlength{\parindent}{0.42cm}
\setlength{\parskip}{0pt}

% ---- Headings: H1 bold, H2/H3 bold italic; number glued to the title ----
\usepackage{titlesec}
\titleformat{\section}       {\normalfont\large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}    {\normalfont\normalsize\bfseries\itshape}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection} {\normalfont\normalsize\bfseries\itshape}{\thesubsubsection}{0.6em}{}
\titlespacing*{\section}      {0pt}{1.3em}{0.6em}
\titlespacing*{\subsection}   {0pt}{1.0em}{0.3em}
\titlespacing*{\subsubsection}{0pt}{0.8em}{0.3em}

% ---- Hypothesis statements: centred, italic, and set off from the body ----
%      \centering (not the center environment) keeps the surrounding vertical
%      space under our own control; \par must precede \endgroup so the final
%      line is still centred.
\newenvironment{hypothesis}
  {\par\addvspace{1.0em}\begingroup\centering\itshape\parindent=0pt}
  {\par\endgroup\addvspace{1.0em}}

% ---- Captions: "Fig. N." / "Table N." with a bold label and a period ----
\usepackage{caption}
\captionsetup{font=small,labelfont=bf,labelsep=period,skip=6pt}
\captionsetup[figure]{name=Fig.}
\captionsetup[table]{name=Table}

% ---- Small Springer-style running head (empty on the title page) ----
\usepackage{fancyhdr}
\providecommand{\runningtitle}{}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\fancyhead[L]{\small\itshape\runningtitle}
\fancyhead[R]{\small\thepage}

% ---- Black links, like the printed template ----
\AtBeginDocument{\hypersetup{hidelinks}}
