% ================= COINs / Springer T1-book look (pdflatex) =================
% NOTE: geometry (margins) is configured in report.qmd via the `geometry:` key,
% and graphicx/hyperref are provided by Quarto, so they are NOT loaded here
% (loading them twice causes an "option clash" error).

% ---- Times body via mathptmx (portable across all TeX Live installs) ----
\usepackage{mathptmx}

% ---- Paragraphs: 0.42 cm first-line indent, no inter-paragraph gap.
%      Requires `indent: true` in report.qmd (otherwise Quarto zeroes the indent).
%      article does not indent the first paragraph after a heading -> matches template.
\setlength{\parindent}{0.42cm}
\setlength{\parskip}{0pt}

% ---- Headings: H1 bold, H2/H3 bold italic; number glued to the title ----
\usepackage{titlesec}
\titleformat{\section}       {\normalfont\large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}    {\normalfont\normalsize\bfseries\itshape}{\thesubsection}{0.6em}{}
\titleformat{\subsubsection} {\normalfont\normalsize\bfseries\itshape}{\thesubsubsection}{0.6em}{}
\titlespacing*{\section}      {0pt}{1.3em}{0.6em}
\titlespacing*{\subsection}   {0pt}{1.0em}{0.3em}
\titlespacing*{\subsubsection}{0pt}{0.8em}{0.3em}

% ---- Float placement: keep every figure inside the section it belongs to ----
%      Without this, LaTeX may defer a float past the next heading, which in a
%      figure-dense chapter such as the appendix pushes images into the wrong
%      section. placeins puts a \FloatBarrier before every \section; the
%      \AtBeginDocument patch extends the same barrier to \subsection and
%      \subsubsection -- the appendix is organised at subsection level and the
%      results chapter at subsubsection level, so both need one. The float
%      parameters below then let a page carry more of its own floats, so
%      barriers rarely have to flush several figures onto a page of their own.
\usepackage[section]{placeins}
\makeatletter
\AtBeginDocument{%
  \expandafter\renewcommand\expandafter\subsection\expandafter{%
    \expandafter\@fb@secFB\subsection}%
  \expandafter\renewcommand\expandafter\subsubsection\expandafter{%
    \expandafter\@fb@secFB\subsubsection}%
}
\makeatother
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{4}
\renewcommand{\topfraction}{0.9}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.7}

% ---- Hypothesis statements: centred, italic, and set off from the body ----
%      \centering (not the center environment) keeps the surrounding vertical
%      space under our own control; \par must precede \endgroup so the final
%      line is still centred.
\newenvironment{hypothesis}
  {\par\addvspace{1.0em}\begingroup\centering\itshape\parindent=0pt}
  {\par\endgroup\addvspace{1.0em}}

% ---- Captions: "Fig. N." / "Table N." with a bold label and a period ----
\usepackage{caption}
\captionsetup{font=small,labelfont=bf,labelsep=period,skip=6pt}
\captionsetup[figure]{name=Fig.}
\captionsetup[table]{name=Table}

% ---- Small Springer-style running head (empty on the title page) ----
\usepackage{fancyhdr}
\providecommand{\runningtitle}{}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.4pt}
\fancyhead[L]{\small\itshape\runningtitle}
\fancyhead[R]{\small\thepage}

% ---- Short List-of-Figures / List-of-Tables entries ----
%      A caption written as "Short Title: full descriptive caption." keeps its
%      full text under the float but contributes only "Short Title" to the
%      list. Quarto emits \caption{\label{fig-x}Short Title: ...} for figures
%      (label inside, first) and \caption{Short Title: ...}\label{tbl-x} for
%      tables, so a leading \label is skipped before the text is split at the
%      first colon. Captions without a colon are passed through unchanged.
%      The patch is deferred to \AtBeginDocument because hyperref and the
%      float packages redefine \caption after this file is read.
\makeatletter
\def\coin@short{}
\def\coin@dosplit#1:#2\@nil{\gdef\coin@short{#1}}
\def\coin@split#1\@nil{\coin@dosplit#1:\@nil}
\def\coin@eatlabel\label#1{\coin@split}
\def\coin@trylabel{\@ifnextchar\label{\coin@eatlabel}{\coin@split}}
\newcommand{\coin@setshort}[1]{\coin@trylabel#1\@nil}
\newcommand{\coin@autocaption}[1]{%
  \coin@setshort{#1}%
  \expandafter\coin@caption\expandafter[\coin@short]{#1}%
}
\AtBeginDocument{%
  \let\coin@caption\caption
  % A caption that already carries an explicit short form, or a starred one,
  % is passed straight through untouched.
  \renewcommand{\caption}{%
    \@ifstar{\coin@caption*}%
            {\@ifnextchar[{\coin@caption}{\coin@autocaption}}%
  }%
}
\makeatother

% ---- Reference list: a smaller gap between entries ----
%      Pandoc opens the bibliography as \begin{CSLReferences}{1}{1}, and its own
%      definition turns that second argument into \itemsep = 1\baselineskip, i.e.
%      a full blank line between entries. That is more air than the printed
%      template uses and pushed the last entry onto a page of its own. The
%      definition below is pandoc's, with the entry spacing scaled down; it is
%      placed here because this file is included after the template has defined
%      the environment, so the redefinition wins. The hanging indent (#1) and
%      \cslhangindent stay untouched.
\renewenvironment{CSLReferences}[2]
 {\begin{list}{}{%
   \setlength{\itemindent}{0pt}%
   \setlength{\leftmargin}{0pt}%
   \setlength{\parsep}{0pt}%
   \ifodd #1
    \setlength{\leftmargin}{\cslhangindent}%
    \setlength{\itemindent}{-1\cslhangindent}%
   \fi
   \setlength{\itemsep}{#2\dimexpr 0.4\baselineskip\relax}}}
 {\end{list}}
