%% partials/title.tex
%% Overrides Quarto's default PDF title block with the left-aligned
%% Springer T1-book opening: big bold title, bold author line, small
%% italic affiliation, and a bold run-in "Abstract" label.
%% Driven by the metadata keys set in report.qmd:
%%   title, group, members (list of name/id), affiliation, email,
%%   runningtitle, paperabstract
%% NOTE: the key must NOT be called `author`/`authors` -- Quarto reserves those
%% and rewrites them into its own author schema, which loses the id field.
%%
%% IMPORTANT: pandoc injects this partial into the PREAMBLE (it is the slot
%% that normally holds \title/\author/\date), so it must contain declarations
%% only -- any text typeset here lands before \begin{document} and aborts the
%% run with "Missing \begin{document}". The visible block therefore goes into
%% \maketitle, which the template calls immediately after \begin{document}.
\gdef\runningtitle{$if(runningtitle)$$runningtitle$$else$$title$$endif$}
\renewcommand{\maketitle}{%
  \pagenumbering{roman}%   front matter counts i, ii, ...; arabic restarts at the Introduction
  \pagestyle{empty}%       no running head anywhere in the front matter; fancy resumes at the Introduction
  \thispagestyle{empty}%
  {\fontsize{16}{18}\selectfont\bfseries\noindent $title$\par}%
  \vspace{1.4em}%
  $if(group)${\noindent\bfseries $group$\par}\vspace{0.4em}$endif$%
  $for(members)$%
  {\noindent\bfseries $members.name$$if(members.id)$\normalfont, $members.id$$endif$\par}%
  $endfor$%
  $if(supervisor)$\vspace{0.4em}{\footnotesize\noindent Supervisor: $supervisor$\par}$endif$%
  $if(cosupervisor)${\footnotesize\noindent Co-Supervisor: $cosupervisor$\par}$endif$%
  $if(affiliation)$\vspace{0.4em}{\footnotesize\itshape\noindent $affiliation$\par}$endif$%
  $if(email)${\footnotesize\noindent $email$\par}$endif$%
  \vspace{1.4em}%
  $if(paperabstract)${\noindent\textbf{Abstract}\quad $paperabstract$\par}$endif$%
  \vspace{1.2em}%
}
